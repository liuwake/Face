cmake_minimum_required(VERSION 3.10)
project(InspireFaceTRTDemo)

# Set C++ standard
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find OpenCV components
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui imgcodecs)

# Define paths
set(INSPIRE_FACE_ROOT "/kaggle/temp/InspireFace")
set(INSPIRE_FACE_LIB_DIR "${INSPIRE_FACE_ROOT}/build/inspireface-linux-tensorrt-cudaV13278_ubuntu_none/InspireFace/lib")
set(TENSORRT_ROOT "/usr" CACHE PATH "Path to TensorRT installation")

# Include directories
include_directories(
    ${OpenCV_INCLUDE_DIRS}
    "${INSPIRE_FACE_ROOT}/build/inspireface-linux-tensorrt-cudaV13278_ubuntu_none/InspireFace/include"
)

# Link directories
link_directories(${INSPIRE_FACE_LIB_DIR})

# Find CUDA
find_package(CUDA REQUIRED)
include_directories(${CUDA_INCLUDE_DIRS})
link_directories(${CUDA_TOOLKIT_ROOT_DIR}/lib64)

# Find TensorRT
find_path(TENSORRT_INCLUDE_DIR NvInfer.h
    HINTS ${TENSORRT_ROOT}
    PATH_SUFFIXES include)

find_library(TENSORRT_LIBRARY_INFER nvinfer
    HINTS ${TENSORRT_ROOT}
    PATH_SUFFIXES lib lib64)

find_library(TENSORRT_LIBRARY_RUNTIME nvinfer_plugin
    HINTS ${TENSORRT_ROOT}
    PATH_SUFFIXES lib lib64)

find_library(CUDA_RUNTIME_LIBRARY cudart
    HINTS ${CUDA_TOOLKIT_ROOT_DIR}
    PATH_SUFFIXES lib64 lib lib64/stubs lib/stubs)

include_directories(${TENSORRT_INCLUDE_DIR})

# Add trt executable
add_executable(trt trt.cpp)

# Define TensorRT macro
target_compile_definitions(trt PRIVATE ISF_ENABLE_TENSORRT)

# Link libraries
target_link_libraries(trt
    ${OpenCV_LIBS}
    InspireFace
    ${TENSORRT_LIBRARY_INFER}
    ${TENSORRT_LIBRARY_RUNTIME}
    ${CUDA_RUNTIME_LIBRARY}
)
